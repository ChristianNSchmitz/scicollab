45quantify_ki <- function(fcs, gates, min_events = 10000) {
46 ff <- flowCore::read.FCS(fcs, transformation = FALSE)
47 stopifnot(nrow(ff) >= min_events)
48
49 # singlets first — order matters, see issue #40
50 live <- apply_gate(ff, gates$singlets) |>
51 apply_gate(gates$viable)
52
53 gfp <- apply_gate(live, gates$gfp_pos)
54 mch <- apply_gate(live, gates$mcherry_pos)
55
56 list(
57 efficiency = nrow(gfp) / nrow(live) * 100,
58 n_live = nrow(live),
59 gate_hash = digest(gates)
60 )
61}
README.md
gate-and-quantify — the gating pipeline behind DS-1174 and MC-0288. Reads an FCS file and a YAML gate definition, returns efficiency, live count and a hash of the gates used, so a number can always be traced to the gate that produced it.
